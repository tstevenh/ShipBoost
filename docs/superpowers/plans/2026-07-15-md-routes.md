# Per-page `.md` Routes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `.md` URL to every tool, best, alternatives, and blog page that returns a clean markdown version of that page for AI crawlers, agents, and developers.

**Architecture:** One new utility file (`src/server/seo/page-markdown.ts`) exports four markdown builder functions (one per page type). Four route handlers (each in a `[slug].md/route.ts` file using the same directory-with-extension pattern as the existing `llms.txt` route) call those builders and return `text/plain`. Each page gets a small developer-facing `.md` link in its UI.

**Tech Stack:** Next.js 16 App Router, TypeScript, Vitest. No new dependencies.

## Global Constraints

- Next.js 16.2.2 — use `params: Promise<{ slug: string }>` (must `await context.params` before use)
- All commands run from `my-app/` directory
- Tests: `npx vitest run <path>` (one-shot); test files sit next to their source files
- Path alias `@/*` → `src/*`
- `revalidate` values must match the corresponding `page.tsx`: tools = `3600`, best/alternatives = `1800`, blog = use `3600` (no override set in blog page)
- `content-type` header: `text/plain; charset=utf-8`
- Cache-control header: `public, max-age=0, s-maxage=<revalidate>, stale-while-revalidate=86400`
- No new npm dependencies

```typescript
export type ToolMarkdownInput = {
  slug: string;
  name: string;
  tagline: string;
  richDescription: string;
  pricingModel: string;
  websiteUrl: string;
  upvoteCount: number;
  toolCategories: Array<{ category: { name: string; slug: string } }>;
  toolTags: Array<{ tag: { name: string; slug: string; isActive: boolean } }>;
  launches: Array<{ launchType: string; launchDate: Date | string }>;
};

export type BestMarkdownInput = {
  entry: {
    slug: string;
    title: string;
    metaDescription: string;
    intro: string;
    rankedTools: Array<{ toolSlug: string; rank: number }>;
  };
  tools: Array<{ slug: string; name: string; tagline: string }>;
};

export type AlternativesMarkdownInput = {
  entry: {
    slug: string;
    title: string;
    intro: string;
    metaDescription: string;
    anchorToolSlug: string;
  };
  anchorTool: { slug: string; name: string; tagline: string; websiteUrl: string };
  tools: Array<{ slug: string; name: string; tagline: string }>;
};

export type BlogMarkdownInput = {
  slug: string;
  title: string;
  excerpt: string;
  markdownContent: string;
  publishedAt: Date | string | null;
  primaryCategory: { name: string };
  author: { name: string };
};

function formatLaunchType(launchType: string): string {
  if (launchType === "FEATURED") return "Featured Launch";
  if (launchType === "FREE") return "Free Launch";
  if (launchType === "RELAUNCH") return "Relaunch";
  return launchType;
}

function formatDate(value: Date | string | null | undefined): string {
  if (!value) return "Unknown";
  return new Intl.DateTimeFormat("en", { dateStyle: "medium" }).format(
    new Date(value),
  );
}

export function buildToolMarkdown(tool: ToolMarkdownInput, appUrl: string): string {
  const categories = tool.toolCategories
    .map((tc) => tc.category.name)
    .join(", ");
  const tags = tool.toolTags
    .filter((tt) => tt.tag.isActive)
    .map((tt) => tt.tag.name)
    .join(", ");
  const listingUrl = `${appUrl}/tools/${tool.slug}`;
  const markdownUrl = `${appUrl}/tools/${tool.slug}.md`;

  const launchLines = tool.launches
    .map((l) => `- ${formatLaunchType(l.launchType)} — ${formatDate(l.launchDate)}`)
    .join("\n");

  return [
    `# ${tool.name}`,
    "",
    `> ${tool.tagline}`,
    "",
    `**Website:** ${tool.websiteUrl}`,
    `**Pricing:** ${tool.pricingModel}`,
    categories ? `**Categories:** ${categories}` : null,
    tags ? `**Tags:** ${tags}` : null,
    `**Upvotes:** ${tool.upvoteCount}`,
    "",
    "## Description",
    "",
    tool.richDescription,
    "",
    ...(tool.launches.length > 0
      ? ["## Launch History", "", launchLines, ""]
      : []),
    "## Links",
    "",
    `- ShipBoost listing: ${listingUrl}`,
    `- Markdown: ${markdownUrl}`,
  ]
    .filter((line) => line !== null)
    .join("\n");
}

export function buildBestMarkdown(page: BestMarkdownInput, appUrl: string): string {
  const { entry, tools } = page;
  const toolsBySlug = new Map(tools.map((t) => [t.slug, t]));
  const sourceUrl = `${appUrl}/best/${entry.slug}`;
  const markdownUrl = `${appUrl}/best/${entry.slug}.md`;

  const toolLines = entry.rankedTools
    .map((ranked) => {
      const tool = toolsBySlug.get(ranked.toolSlug);
      if (!tool) return null;
      return `${ranked.rank}. **${tool.name}** — ${tool.tagline} — ${appUrl}/tools/${tool.slug}`;
    })
    .filter(Boolean)
    .join("\n");

  return [
    `# ${entry.title}`,
    "",
    entry.metaDescription,
    "",
    entry.intro,
    "",
    "## Ranked Tools",
    "",
    toolLines,
    "",
    "---",
    "",
    `Source: ${sourceUrl}`,
    `Markdown: ${markdownUrl}`,
  ].join("\n");
}

export function buildAlternativesMarkdown(
  page: AlternativesMarkdownInput,
  appUrl: string,
): string {
  const { entry, anchorTool, tools } = page;
  const sourceUrl = `${appUrl}/alternatives/${entry.slug}`;
  const markdownUrl = `${appUrl}/alternatives/${entry.slug}.md`;

  const altLines = tools
    .map((t) => `- **${t.name}** — ${t.tagline} — ${appUrl}/tools/${t.slug}`)
    .join("\n");

  return [
    `# ${entry.title}`,
    "",
    entry.metaDescription,
    "",
    entry.intro,
    "",
    `## ${anchorTool.name}`,
    "",
    anchorTool.tagline,
    `Website: ${anchorTool.websiteUrl}`,
    `ShipBoost: ${appUrl}/tools/${anchorTool.slug}`,
    "",
    "## Alternatives",
    "",
    altLines,
    "",
    "---",
    "",
    `Source: ${sourceUrl}`,
    `Markdown: ${markdownUrl}`,
  ].join("\n");
}

export function buildBlogMarkdown(article: BlogMarkdownInput, appUrl: string): string {
  const sourceUrl = `${appUrl}/blog/${article.slug}`;
  const markdownUrl = `${appUrl}/blog/${article.slug}.md`;
  const dateStr = formatDate(article.publishedAt);

  return [
    `# ${article.title}`,
    "",
    `By ${article.author.name} · ${dateStr} · ${article.primaryCategory.name}`,
    "",
    `> ${article.excerpt}`,
    "",
    "---",
    "",
    article.markdownContent,
    "",
    "---",
    "",
    `Source: ${sourceUrl}`,
    `Markdown: ${markdownUrl}`,
  ].join("\n");
}
```

```typescript
import { notFound } from "next/navigation";

import {
  getCachedToolPage,
  getCachedToolStaticParams,
} from "@/server/cache/public-content";
import { getEnv } from "@/server/env";
import { buildToolMarkdown } from "@/server/seo/page-markdown";

export const revalidate = 3600;

type RouteContext = {
  params: Promise<{ slug: string }>;
};

export async function generateStaticParams() {
  return getCachedToolStaticParams();
}

export async function GET(_request: Request, context: RouteContext) {
  const { slug } = await context.params;
  const toolPage = await getCachedToolPage(slug);

  if (!toolPage || toolPage.kind !== "public") {
    return notFound();
  }

  const markdown = buildToolMarkdown(toolPage.tool, getEnv().NEXT_PUBLIC_APP_URL);

  return new Response(markdown, {
    headers: {
      "content-type": "text/plain; charset=utf-8",
      "cache-control": "public, max-age=0, s-maxage=3600, stale-while-revalidate=86400",
    },
  });
}
```

- [ ] **Step 1: Create the route handler**

Create `src/app/blog/[slug].md/route.ts`:

```typescript
import { notFound } from "next/navigation";

import {
  getCachedBlogArticleStaticParams,
  getCachedPublishedBlogArticle,
} from "@/server/cache/public-content";
import { getEnv } from "@/server/env";
import { buildBlogMarkdown } from "@/server/seo/page-markdown";

export const revalidate = 3600;

type RouteContext = {
  params: Promise<{ slug: string }>;
};

export async function generateStaticParams() {
  return getCachedBlogArticleStaticParams();
}

export async function GET(_request: Request, context: RouteContext) {
  const { slug } = await context.params;
  const article = await getCachedPublishedBlogArticle(slug);

  if (!article) {
    return notFound();
  }

  const markdown = buildBlogMarkdown(article, getEnv().NEXT_PUBLIC_APP_URL);

  return new Response(markdown, {
    headers: {
      "content-type": "text/plain; charset=utf-8",
      "cache-control": "public, max-age=0, s-maxage=3600, stale-while-revalidate=86400",
    },
  });
}
```
